// --- biquote orqali real vaqtda narxni olish --- const priceDisplay = document.getElementById('livePrice'); async function fetchGoldPrice() { try { // biquote API - API key kerak emas const response = await fetch('https://api.biquote.io/v1/quote?symbol=XAUUSD'); if (!response.ok) throw new Error('Network response was not ok'); const data = await response.json(); // biquote qaytadigan format: { mid: 1234.56, ... } yoki { price: ... } let price = data.mid || data.price || data.ask; if (price) { priceDisplay.textContent = Number(price).toFixed(5); priceDisplay.classList.remove('price-flash'); void priceDisplay.offsetWidth; // animatsiyani qayta ishga tushirish priceDisplay.classList.add('price-flash'); } } catch (error) { console.warn('Narxni yangilab bo\'lmadi, statik qiymat qoldirildi:', error.message); // Xatolik bo'lsa, hech narsa qilmaymiz, eski narx qoladi } } // Sahifa yuklanganda va har 1 daqiqada (60 soniya) narxni yangilash fetchGoldPrice(); setInterval(fetchGoldPrice, 60000);
